-- Kiểm tra Key đã lưu còn hiệu lực (trong vòng 24 giờ) hay không
local function checkSavedKey()
    if isfile and readfile and isfile(fileName) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(fileName))
        end)
        
        if success and data and data.key == correctKey and data.timestamp then
            local currentTime = os.time()
            local timePassed = currentTime - data.timestamp
            -- 24 giờ = 86400 giây
            if timePassed < 86400 then
                return true
            end
        end
    end
    return false
end

-- Hàm lưu Key cùng mốc thời gian
local function saveKey(keyToSave)
    if writefile and makefolder and isfolder then
        if not isfolder(folderName) then
            makefolder(folderName)
        end
        local data = {
            key = keyToSave,
            timestamp = os.time()
        }
        pcall(function()
            writefile(fileName, HttpService:JSONEncode(data))
        end)
    end
end

-- Hiệu ứng Lắc Mượt
local isShaking = false
local function smoothShake()
    if isShaking then return end
    isShaking = true
    
    local originalPos = UDim2.new(0.5, -195, 0.5, -115)
    local duration = 0.45
    local frequency = 35
    local amplitude = 10
    
    local startTime = tick()
    local connection
    connection = RunService.RenderStepped:Connect(function()
        local elapsed = tick() - startTime
        if elapsed >= duration then
            MainFrame.Position = originalPos
            connection:Disconnect()
            isShaking = false
        else
            local damp = 1 - (elapsed / duration)
            local offset = math.sin(elapsed * frequency) * amplitude * damp
            MainFrame.Position = UDim2.new(0.5, -195 + offset, 0.5, -115)
        end
    end)
end

-- Cập nhật thông báo Status Banner
local function updateStatus(text, textColor, bannerColor)
    StatusText.Text = text
    TweenService:Create(StatusText, TweenInfo.new(0.2), {TextColor3 = textColor}):Play()
    TweenService:Create(StatusBanner, TweenInfo.new(0.2), {BackgroundColor3 = bannerColor}):Play()
end

        TextBox.Text = "" -- Tự động xóa chữ
        smoothShake() -- Hiệu ứng lắc mượt
    end
end)
